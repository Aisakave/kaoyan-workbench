// ---- 错题复习（F-08）：自适应间隔与到期计算 ----
const REVIEW_INTERVALS = [3, 5, 7, 10, 14, 21, 30];

// 成功复习 reviewCount 次后的下一次间隔（天），封顶 30
function nextReviewInterval(reviewCount) {
  return REVIEW_INTERVALS[Math.min(reviewCount || 0, REVIEW_INTERVALS.length - 1)];
}

// 到期时间戳：lastReviewedAt + 间隔；从未复习按 created + 3 天
function reviewDue(w) {
  const base = w.lastReviewedAt || w.created || Date.now();
  return base + nextReviewInterval(w.reviewCount) * 86400000;
}

// 复习元信息：due/逾期天数/距上次复习天数/距到期天数
function reviewMeta(w) {
  const now = Date.now();
  const due = reviewDue(w);
  return {
    due,
    overdueDays: Math.max(0, Math.floor((now - due) / 86400000)),
    daysSince: w.lastReviewedAt ? Math.max(0, Math.floor((now - w.lastReviewedAt) / 86400000)) : null,
    daysToDue: Math.max(0, Math.ceil((due - now) / 86400000))
  };
}

// 逾期等级：''=未到期 / due=到期(黄) / warn=逾期3-6天(橙) / danger=逾期≥7天(红)
function overdueLevel(overdueDays) {
  if (overdueDays >= 7) return 'danger';
  if (overdueDays >= 3) return 'warn';
  if (overdueDays >= 1) return 'due';
  return '';
}
```

  // ---- 错题复习与排行榜（F-08）----
  // 复习=重做：correct=true 做对→reviewCount+1；false 做错→recurCount+1、reviewCount 归零、≥3 永久保留
  function markReviewed(id, correct) {
    const w = state.wrongQuestions.find(x => x.id === id);
    if (!w) return null;
    w.lastReviewedAt = Date.now();
    if (correct) {
      w.reviewCount = (w.reviewCount || 0) + 1;
    } else {
      w.recurCount = (w.recurCount || 0) + 1;
      w.reviewCount = 0;
      if (w.recurCount >= 3) w.permanent = true;
    }
    w.updated = Date.now();
    persist();
    return { recurCount: w.recurCount, reviewCount: w.reviewCount, permanent: w.permanent };
  }

  // 到期未复习队列：按到期时间升序
  function reviewDueList() {
    const now = Date.now();
    return state.wrongQuestions
      .map(w => ({ w, meta: reviewMeta(w) }))
      .filter(x => x.meta.due <= now)
      .sort((a, b) => a.meta.due - b.meta.due);
  }

  function reviewDueCount() { return reviewDueList().length; }

  // 排行榜：按复发次数降序，次级按最近更新降序
  function ranking(limit = 10) {
    return state.wrongQuestions.slice()
      .sort((a, b) => (b.recurCount || 0) - (a.recurCount || 0) || (b.updated || 0) - (a.updated || 0))
      .slice(0, limit);
  }
```
